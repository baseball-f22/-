    <script>
        // 初始化本機備份
        window.athletes = JSON.parse(localStorage.getItem('rapsodo_athletes')) || {};

        function saveDataLocally() {
            localStorage.setItem('rapsodo_athletes', JSON.stringify(window.athletes));
            localStorage.setItem('rapsodo_records', JSON.stringify(window.records));

            // 同步上傳至雲端資料庫
            if (window.rapsodoRecordsRef && window.set) window.set(window.rapsodoRecordsRef, window.records);
            if (window.rapsodoAthletesRef && window.set) window.set(window.rapsodoAthletesRef, window.athletes);
        }

        // 匯入 Rapsodo CSV 邏輯
        function importCSV(e) {
            const file = e.target.files[0];
            if (!file) return;

            Papa.parse(file, {
                header: true,
                skipEmptyLines: true,
                complete: function(results) {
                    let count = 0;
                    results.data.forEach(row => {
                        const pitchId = row['PitchID'];
                        const speed = parseFloat(row['RelSpeed (MPH)']);
                        const pitchType = row['Pitch Type'];
                        
                        if (!pitchId || isNaN(speed) || !pitchType) return;
                        if (window.records.some(r => r.pitchId === pitchId)) return; // 避免重複匯入同一顆球
                        
                        const rawDate = row['Pitch Created At'];
                        let dateStr = rawDate;
                        if (rawDate) {
                            const parts = rawDate.split(' ');
                            if (parts.length > 0) {
                                const dParts = parts[0].split('-');
                                if (dParts.length === 3 && dParts[0].length === 2 && dParts[2].length === 4) {
                                    dateStr = `${dParts[2]}-${dParts[0]}-${dParts[1]}`;
                                } else {
                                    dateStr = parts[0];
                                }
                            }
                        }
                        
                        let name = row['Pitcher First Name'] ? row['Pitcher First Name'].trim() : "未命名投手";
                        if (!window.athletes[name]) {
                            window.athletes[name] = { gender: '男', dob: '', height: '', weight: '' };
                        }
                        
                        window.records.push({
                            pitchId: pitchId,
                            name: name,
                            date: dateStr,
                            pitchType: pitchType,
                            speed: speed
                        });
                        count++;
                    });
                    
                    alert(`成功匯入 ${count} 筆新投球數據（舊資料已完整保留並自動雲端同步）！`);
                    saveDataLocally();
                    updateDropdowns();
                    updateDashboard();
                    e.target.value = '';
                }
            });
        }

        function getPitchTypeName(pt) {
            const map = {
                'FB': '快速球', 'FA': '快速球', 'FF': '四縫線', 'FT': '二縫線', 'SI': '伸卡球',
                'CB': '曲球', 'CU': '曲球', 'SL': '滑球', 'CH': '變速球',
                'SP': '指叉球', 'FS': '指叉球', 'FC': '卡特球'
            };
            return map[pt] || '未知';
        }

        function updateDropdowns() {
            const athSelect = document.getElementById('athleteSelect');
            const currAth = athSelect.value;
            
            const names = [...new Set([...Object.keys(window.athletes), ...window.records.map(r => r.name)])];
            athSelect.innerHTML = names.length ? '' : '<option value="">請先匯入資料</option>';
            names.forEach(n => {
                const opt = document.createElement('option'); opt.value = n; opt.textContent = n;
                athSelect.appendChild(opt);
            });
            
            if (names.includes(currAth)) athSelect.value = currAth;
            else if (names.length > 0) athSelect.value = names[0];
            
            onAthleteChange();
        }

        function onAthleteChange() {
            const ptSelect = document.getElementById('pitchTypeSelect');
            const selectedAth = document.getElementById('athleteSelect').value;
            
            if (!selectedAth) {
                ptSelect.innerHTML = '<option value="">無資料</option>';
                return;
            }

            const athRecords = window.records.filter(r => r.name === selectedAth);
            const pTypes = [...new Set(athRecords.map(r => r.pitchType))];
            
            const ptCounts = {};
            athRecords.forEach(r => { ptCounts[r.pitchType] = (ptCounts[r.pitchType] || 0) + 1; });
            pTypes.sort((a, b) => ptCounts[b] - ptCounts[a]);

            ptSelect.innerHTML = pTypes.length ? '' : '<option value="">無資料</option>';
            pTypes.forEach(pt => {
                const opt = document.createElement('option');
                opt.value = pt; 
                opt.textContent = `${pt} ${getPitchTypeName(pt)} (共 ${ptCounts[pt]} 球)`;
                ptSelect.appendChild(opt);
            });

            updateDashboard();
        }

        function getStabilityDiagnosis(sd, count) {
            if (count < 2) return "球數不足，無法計算標準差";
            if (sd <= 1.0) return "極佳 (放球點與發力非常固定)";
            if (sd <= 1.8) return "良好 (具備實戰基本穩定度)";
            if (sd <= 2.5) return "尚可 (偶有失投與球速落差)";
            return "不穩定 (需教練協助調整發力機制)";
        }

        function getStabilityColor(sd, count) {
            if (count < 2) return "text-slate-400";
            if (sd <= 1.0) return "text-emerald-400";
            if (sd <= 1.8) return "text-yellow-400";
            return "text-red-400";
        }

        function updateDashboard() {
            const ath = document.getElementById('athleteSelect').value;
            const pt = document.getElementById('pitchTypeSelect').value;
            
            if (!ath || !pt) return;
            
            const profileDiv = document.getElementById('athleteProfile');
            const p = window.athletes[ath] || {};
            profileDiv.classList.remove('hidden');
            profileDiv.innerHTML = `
                <div class="font-bold text-sky-300 text-base border-r border-slate-600 pr-3">${ath}</div>
                <div>性別: <span class="text-white">${p.gender||'--'}</span></div>
                <div>生日: <span class="text-white">${p.dob||'--'}</span></div>
                <div>身高: <span class="text-white">${p.height?p.height+' cm':'--'}</span></div>
                <div>體重: <span class="text-white">${p.weight?p.weight+' kg':'--'}</span></div>
            `;
            
            const filtered = window.records.filter(r => r.name === ath && r.pitchType === pt);
            const grouped = {};
            filtered.forEach(r => {
                if (!grouped[r.date]) grouped[r.date] = [];
                grouped[r.date].push(r.speed);
            });
            
            const dates = Object.keys(grouped).sort();
            const dailyStats = dates.map(date => {
                const speeds = grouped[date];
                const count = speeds.length;
                const max = Math.max(...speeds);
                const avg = speeds.reduce((a,b)=>a+b,0) / count;
                
                let sd = 0;
                if (count > 1) {
                    const variance = speeds.reduce((acc, val) => acc + Math.pow(val - avg, 2), 0) / (count - 1);
                    sd = Math.sqrt(variance);
                }
                return { date, count, max, avg, sd };
            });
            
            if (dailyStats.length > 0) {
                const latest = dailyStats[dailyStats.length - 1];
                document.getElementById('kpiDate').textContent = latest.date;
                document.getElementById('kpiCount').textContent = latest.count + ' 球';
                document.getElementById('kpiMax').textContent = latest.max.toFixed(1);
                document.getElementById('kpiAvg').textContent = latest.avg.toFixed(1);
                document.getElementById('kpiSD').textContent = latest.count > 1 ? latest.sd.toFixed(2) : '--';
                
                const evalEl = document.getElementById('kpiEval');
                evalEl.textContent = getStabilityDiagnosis(latest.sd, latest.count);
                evalEl.className = `text-sm md:text-md font-bold mt-1 leading-tight ${getStabilityColor(latest.sd, latest.count)}`;
            }
            
            const tbody = document.getElementById('dataTableBody');
            tbody.innerHTML = '';
            [...dailyStats].reverse().forEach(stat => {
                tbody.innerHTML += `
                    <tr class="hover:bg-slate-800/30">
                        <td class="p-3">${stat.date}</td>
                        <td class="p-3 text-sky-300 font-semibold">${stat.count}</td>
                        <td class="p-3 text-emerald-400 font-semibold">${stat.max.toFixed(1)}</td>
                        <td class="p-3 text-amber-400 font-semibold">${stat.avg.toFixed(1)}</td>
                        <td class="p-3 text-purple-400 font-bold">${stat.count > 1 ? stat.sd.toFixed(2) : '--'}</td>
                        <td class="p-3 text-xs font-medium ${getStabilityColor(stat.sd, stat.count)}">${getStabilityDiagnosis(stat.sd, stat.count)}</td>
                    </tr>
                `;
            });
            
            renderCharts(dailyStats);
        }

        function renderCharts(stats) {
            const dates = stats.map(s => s.date);
            const commonLayout = {
                paper_bgcolor: 'transparent', plot_bgcolor: 'transparent', font: { color: '#94a3b8' },

            };
            
            const traceMax = { x: dates, y: stats.map(s => s.max), name: '極速 (Max)', type: 'scatter', mode: 'lines+markers', line: { color: '#34d399', width: 2, dash: 'dot' }, marker: { size: 6 } };
            const traceAvg = { x: dates, y: stats.map(s => s.avg), name: '均速 (Avg)', type: 'scatter', mode: 'lines+markers', line: { color: '#38bdf8', width: 3 }, marker: { size: 8 } };
            
            Plotly.newPlot('speedChart', [traceMax, traceAvg], { ...commonLayout, yaxis: { title: '球速 (MPH)', gridcolor: '#334155' }, legend: { orientation: 'h', y: 1.2, x: 0 } }, { responsive: true, displayModeBar: false });
            
            const sdColors = stats.map(s => s.count < 2 ? 'transparent' : (s.sd <= 1.0 ? '#10b981' : (s.sd <= 1.8 ? '#f59e0b' : '#ef4444')));
            const traceSD = { x: dates, y: stats.map(s => s.sd), name: '標準差', type: 'bar', marker: { color: sdColors }, text: stats.map(s => s.count > 1 ? s.sd.toFixed(2) : ''), textposition: 'auto' };
            
            Plotly.newPlot('sdChart', [traceSD], { ...commonLayout, yaxis: { title: '標準差 (SD)', gridcolor: '#334155' } }, { responsive: true, displayModeBar: false });
        }

        function openAthleteModal() {
            const ath = document.getElementById('athleteSelect').value;
            if (!ath) return alert('請先選擇或匯入投手資料。');
            const profile = window.athletes[ath] || {};
            document.getElementById('athleteModalName').value = ath;
            document.getElementById('athleteModalGender').value = profile.gender || '男';
            document.getElementById('athleteModalDOB').value = profile.dob || '';
            document.getElementById('athleteModalHeight').value = profile.height || '';
            document.getElementById('athleteModalWeight').value = profile.weight || '';
            document.getElementById('athleteModal').classList.remove('hidden');
        }

        function closeAthleteModal() { document.getElementById('athleteModal').classList.add('hidden'); }

        function handleAthleteSubmit(e) {
            e.preventDefault();
            const ath = document.getElementById('athleteModalName').value;
            window.athletes[ath] = {
                gender: document.getElementById('athleteModalGender').value,
                dob: document.getElementById('athleteModalDOB').value,
                height: parseFloat(document.getElementById('athleteModalHeight').value) || '',
                weight: parseFloat(document.getElementById('athleteModalWeight').value) || ''
            };
            saveDataLocally();
            closeAthleteModal();
            updateDashboard();
        }
